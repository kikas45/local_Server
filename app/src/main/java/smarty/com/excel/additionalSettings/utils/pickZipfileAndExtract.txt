package sync2app.com.syncapplive.additionalSettings

import android.content.Intent
import android.media.MediaScannerConnection
import android.net.Uri
import android.os.Bundle
import android.util.Log
import android.widget.Toast
import androidx.activity.result.contract.ActivityResultContracts
import androidx.appcompat.app.AppCompatActivity
import androidx.lifecycle.lifecycleScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext
import sync2app.com.syncapplive.additionalSettings.utils.Constants
import sync2app.com.syncapplive.databinding.ActivitySampleDemoV2Binding
import java.io.File
import java.io.FileOutputStream
import java.io.InputStream
import java.util.zip.ZipEntry
import java.util.zip.ZipInputStream

class DE_MO_202100_2222 : AppCompatActivity() {


    private lateinit var binding: ActivitySampleDemoV2Binding

    val getFolderClo = "CLO"
    val getFolderSubpath = "USER"
    val Zip = "Zip"
    val fileName = "App.zip"



    private var selectedZipUri: Uri? = null
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        binding = ActivitySampleDemoV2Binding.inflate(layoutInflater)
        setContentView(binding.root)




        binding.textGoBack.setOnClickListener {
            startActivity(Intent(this, SelectDownloadUrlActivity::class.java))
            finish()
        }


        binding.textExtractZip.setOnClickListener {
            funUnZipFile()
        }


        binding.textPickZipFileFolderFromStorage.setOnClickListener {
            val intent = Intent(Intent.ACTION_OPEN_DOCUMENT).apply {
                addCategory(Intent.CATEGORY_OPENABLE)
                type = "application/zip"   // Only ZIP files
            }
            pickZipFileLauncher.launch(intent)
        }




    }

    private val pickZipFileLauncher = registerForActivityResult(
        ActivityResultContracts.StartActivityForResult()
    ) { result ->
        if (result.resultCode == RESULT_OK) {
            result.data?.data?.let { uri ->
                selectedZipUri = uri
                binding.textDisplayStatusOfFile.text = "Selected: $uri"
            }
        }
    }




    private suspend fun extractZip(inputStream: InputStream, destinationPath: String) {
        withContext(Dispatchers.Main) {
            binding.textDisplayStatusOfFile.text = Constants.Extracting
        }

        try {
            val buffer = ByteArray(1024)
            val zipInputStream = ZipInputStream(inputStream)

            var entry: ZipEntry? = zipInputStream.nextEntry
            while (entry != null) {
                val entryFile = File(destinationPath, entry.name)
                if (entry.isDirectory) {
                    entryFile.mkdirs()
                } else {
                    val parentDir = entryFile.parentFile
                    if (!parentDir.exists()) parentDir.mkdirs()

                    FileOutputStream(entryFile).use { outputStream ->
                        var len: Int
                        while (zipInputStream.read(buffer).also { len = it } > 0) {
                            outputStream.write(buffer, 0, len)
                        }
                    }
                }

                MediaScannerConnection.scanFile(
                    applicationContext,
                    arrayOf(entryFile.absolutePath),
                    null
                ) { path, uri ->
                    Log.d("MediaScanner", "Scanned $path -> $uri")
                    runOnUiThread {
                        binding.textDisplayStatusOfFile.text = "$uri"
                    }
                }

                entry = zipInputStream.nextEntry
            }

            zipInputStream.close()

            withContext(Dispatchers.Main) {
                showToastMessage("Extraction completed successfully.")
                allExtractionCompleted()
            }
        } catch (e: Exception) {
            e.printStackTrace()
            withContext(Dispatchers.Main) {
                showToastMessage("Error during extraction: ${e.localizedMessage}")
                allExtractionCompleted()
            }
        }
    }




    private fun funUnZipFile() {
        val uri = selectedZipUri
        if (uri == null) {
            showToastMessage("Please select a ZIP file first.")
            return
        }

        lifecycleScope.launch(Dispatchers.IO) {
            try {
                val baseDir = getExternalFilesDir(null)
                val extractToDir = File(baseDir, "${Constants.Syn2AppLive}/$getFolderClo/$getFolderSubpath/${Constants.App}")
                if (!extractToDir.exists()) extractToDir.mkdirs()

                contentResolver.openInputStream(uri)?.use { inputStream ->
                    extractZip(inputStream, extractToDir.absolutePath)
                }

            } catch (e: Exception) {
                e.printStackTrace()
                withContext(Dispatchers.Main) {
                    showToastMessage("Error: ${e.localizedMessage}")
                }
            }
        }
    }










    private fun allExtractionCompleted() {
        binding.textDisplayStatusOfFile.text = "Completed"
        showToastMessage(Constants.media_ready)

        lifecycleScope.launch {
          //  startTheActivity()
        }

    }


    private fun showToastMessage(messages: String) {

        try {
            Toast.makeText(applicationContext, messages, Toast.LENGTH_SHORT).show()
        } catch (_: Exception) {
        }
    }

}
